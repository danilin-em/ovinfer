# ovinfer v0.1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Реализовать ovinfer v0.1 — Python-пакет с синхронным ядром инференса (LRU-кеш пайплайнов, device-map, preload, LLM chat со стримингом, embeddings) и OpenAI-совместимым FastAPI-транспортом, запускаемым как `ovinfer server`.

**Architecture:** Sync-ядро (`backend.py` + `config.py`/`errors.py`/`registry.py`) не импортирует FastAPI; тонкий транспорт (`server.py`) маппит доменные результаты/ошибки на OpenAI-формат, `def`-роуты выполняются в threadpool. Перенесённый каркас `server.py` из корня растворяется в пакет и удаляется.

**Tech Stack:** Python ≥3.12, uv, openvino==2026.2.0, openvino-genai==2026.2.0, FastAPI, uvicorn, python-multipart.

class OvinferError(Exception):
    """Base class for ovinfer domain errors."""


class ModelNotFound(OvinferError):
    """The OpenAI `model` field does not resolve to a local model directory."""

        super().__init__(f"Model '{model_id}' not found in {models_root}")

    """The endpoint exists in the contract but its pipeline is not wired yet."""

        super().__init__(f"{pipeline} backend is not implemented yet")

```python
"""Settings resolution: explicit arguments > OVINFER_* env > defaults.

The OVINFER_ prefix is deliberate: the short OV_ prefix is taken by
OpenVINO's own environment variables.
"""

import os
from dataclasses import dataclass
from pathlib import Path

from .errors import ConfigError

_ENV_PREFIX = "OVINFER_"
PIPELINE_KINDS = ("llm", "embedding")


def _xdg_data_home() -> Path:
    raw = os.environ.get("XDG_DATA_HOME", "")
    return Path(raw) if raw else Path.home() / ".local" / "share"


def default_models_root() -> Path:
    return _xdg_data_home() / "ovinfer" / "models"


def parse_device_map(raw: str) -> dict[str, str]:
    """Parse 'org/model=NPU,org/model2=GPU' into {model_id: device}."""
    device_map: dict[str, str] = {}
    for entry in filter(None, (e.strip() for e in raw.split(","))):
        model_id, sep, device = entry.partition("=")
        if not sep or not model_id.strip() or not device.strip():
            raise ConfigError(
                f"Invalid device-map entry {entry!r} (expected MODEL=DEVICE)"
            )
        device_map[model_id.strip()] = device.strip()
    return device_map


def parse_preload(raw: str) -> list[tuple[str, str]]:
    """Parse 'org/model[:kind],...' into (model_id, kind); kind defaults to 'llm'."""
    entries: list[tuple[str, str]] = []
    for item in filter(None, (e.strip() for e in raw.split(","))):
        model_id, sep, kind = item.partition(":")
        kind = kind.strip() if sep else "llm"
        if not model_id.strip() or kind not in PIPELINE_KINDS:
            raise ConfigError(
                f"Invalid preload entry {item!r} "
                f"(expected MODEL[:kind], kind one of {PIPELINE_KINDS})"
            )
        entries.append((model_id.strip(), kind))
    return entries


@dataclass(frozen=True)
class Settings:
    models_root: Path
    default_device: str
    device_map: dict[str, str]
    preload: list[tuple[str, str]]
    max_pipelines: int
    use_continuous_batching: bool


def load_settings(
    *,
    models_root: Path | str | None = None,
    default_device: str | None = None,
    device_map: dict[str, str] | None = None,
    preload: list[tuple[str, str]] | None = None,
    max_pipelines: int | None = None,
    use_continuous_batching: bool | None = None,
) -> Settings:
    """Resolve effective settings; raises ConfigError on invalid combinations."""
    env = os.environ
    if models_root is None:
        models_root = env.get(_ENV_PREFIX + "MODELS_ROOT") or default_models_root()
    if default_device is None:
        default_device = env.get(_ENV_PREFIX + "DEVICE", "CPU")
    if device_map is None:
        device_map = parse_device_map(env.get(_ENV_PREFIX + "DEVICE_MAP", ""))
    if preload is None:
        preload = parse_preload(env.get(_ENV_PREFIX + "PRELOAD", ""))
    if use_continuous_batching is None:
        use_continuous_batching = (
            env.get(_ENV_PREFIX + "USE_CONTINUOUS_BATCHING", "0") == "1"
        )
    if max_pipelines is None:
        raw_max = env.get(_ENV_PREFIX + "MAX_PIPELINES")
        if raw_max is None:
            max_pipelines = max(2, len(preload))
        else:
            try:
                max_pipelines = int(raw_max)
            except ValueError:
                raise ConfigError(
                    f"OVINFER_MAX_PIPELINES must be an integer, got {raw_max!r}"
                ) from None
    if max_pipelines < 1:
        raise ConfigError(f"max_pipelines must be >= 1, got {max_pipelines}")
    if len(preload) > max_pipelines:
        raise ConfigError(
            f"{len(preload)} preload models exceed max_pipelines={max_pipelines}; "
            "raise the limit or trim the preload list"
        )
    return Settings(
        models_root=Path(models_root).expanduser(),
        default_device=default_device,
        device_map=dict(device_map),
        preload=list(preload),
        max_pipelines=max_pipelines,
        use_continuous_batching=use_continuous_batching,
    )
```

```python
"""Model registry: scan models_root (org/model layout) and resolve model ids."""

from pathlib import Path

from .errors import ModelNotFound


def list_models(models_root: Path) -> list[str]:
    """Scan models_root two levels deep (org/model layout).

    A directory counts as a model if it contains at least one OpenVINO IR
    (*.xml) anywhere inside (diffusion models keep IRs in subdirectories).
    """
    models: list[str] = []
    if not models_root.is_dir():
        return models
    for org in sorted(models_root.iterdir()):
        if not org.is_dir():
            continue
        for model in sorted(org.iterdir()):
            if model.is_dir() and next(model.rglob("*.xml"), None) is not None:
                models.append(f"{org.name}/{model.name}")
    return models


def resolve_model_dir(models_root: Path, model_id: str) -> Path:
    """Map the OpenAI `model` field to a local directory or raise ModelNotFound.

    Membership check against list_models() also blocks path traversal.
    """
    if model_id not in list_models(models_root):
        raise ModelNotFound(model_id, models_root)
    return models_root / model_id


def model_created(models_root: Path, model_id: str) -> int:
    """Best-effort mtime; 0 if the dir vanished between scan and stat."""
    try:
        return int((models_root / model_id).stat().st_mtime)
    except OSError:
        return 0
```

### Task 4: `backend.py` — каркас ядра: результаты, кеш с локами, заглушки

Самый важный инвариант: два уровня локов. Глобальный `_cache_lock` держится только на время операций со словарём (lookup/insert/evict); создание пайплайна и генерация идут под per-pipeline-локом **вне** глобального. `ModelNotFound` проверяется до касания кеша — 404-запросы не должны вставлять записи и вытеснять живые пайплайны.

**Files:**
- Create: `ovinfer/backend.py`

- [ ] **Step 1: Создать `ovinfer/backend.py`**

```python
"""Synchronous inference core: pipeline cache and generation methods.

All openvino-genai calls live behind Backend. The transport
(ovinfer.server) maps results and domain errors onto the OpenAI wire
format; nothing in this module imports FastAPI. openvino_genai itself is
imported lazily inside methods: importing ovinfer.backend stays cheap and
works without a configured OpenVINO environment.
"""

import queue
import threading
from collections import OrderedDict
from contextlib import contextmanager
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Iterator

from . import config, registry
from .errors import PipelineNotImplemented

_IMAGE_PART_TYPES = {"image_url", "input_image"}
_QUEUE_END = object()


@dataclass(frozen=True)
class ChatResult:
    text: str
    finish_reason: str  # "stop" | "length"
    prompt_tokens: int
    completion_tokens: int


@dataclass(frozen=True)
class EmbedResult:
    vectors: list[list[float]]
    prompt_tokens: int


def messages_have_images(messages: list[dict[str, Any]]) -> bool:
    """True if any message carries an image content part (VLM request)."""
    for msg in messages:
        content = msg.get("content")
        if isinstance(content, list):
            for part in content:
                if part.get("type") in _IMAGE_PART_TYPES:
                    return True
    return False


def _flatten_content(content: str | list[dict[str, Any]] | None) -> str:
    """Join text parts of an OpenAI message content into one string."""
    if content is None:
        return ""
    if isinstance(content, str):
        return content
    return "".join(
        part.get("text", "") for part in content if part.get("type") == "text"
    )


class _CacheEntry:
    __slots__ = ("pipeline", "lock")

    def __init__(self) -> None:
        self.pipeline: Any = None
        self.lock = threading.Lock()


class Backend:
    """Pipeline cache plus one method per contract endpoint.

    Unimplemented endpoints keep plain-Python signatures and raise
    PipelineNotImplemented so the 501 contract holds.
    """

    def __init__(
        self,
        *,
        models_root: Path | str | None = None,
        default_device: str | None = None,
        device_map: dict[str, str] | None = None,
        preload: list[tuple[str, str]] | None = None,
        max_pipelines: int | None = None,
        use_continuous_batching: bool | None = None,
    ) -> None:
        self.settings = config.load_settings(
            models_root=models_root,
            default_device=default_device,
            device_map=device_map,
            preload=preload,
            max_pipelines=max_pipelines,
            use_continuous_batching=use_continuous_batching,
        )
        self._cache: OrderedDict[tuple[str, str], _CacheEntry] = OrderedDict()
        self._cache_lock = threading.Lock()

    # -- registry ------------------------------------------------------------

    def list_models(self) -> list[str]:
        return registry.list_models(self.settings.models_root)

    def model_created(self, model_id: str) -> int:
        return registry.model_created(self.settings.models_root, model_id)

    # -- pipeline cache --------------------------------------------------------

    def preload_models(self) -> None:
        """Eagerly create pipelines listed in settings.preload."""
        for model_id, kind in self.settings.preload:
            with self._pipeline(model_id, kind):
                pass

    def _device_for(self, model_id: str) -> str:
        return self.settings.device_map.get(model_id, self.settings.default_device)

    def _entry(self, model_id: str, kind: str) -> _CacheEntry:
        """Look up or insert a cache entry; evict the LRU one over capacity."""
        key = (model_id, kind)
        with self._cache_lock:
            entry = self._cache.get(key)
            if entry is not None:
                self._cache.move_to_end(key)
                return entry
            entry = _CacheEntry()
            self._cache[key] = entry
            while len(self._cache) > self.settings.max_pipelines:
                # Drop the least recently used entry. The pipeline object is
                # freed by GC once the last thread holding its lock finishes;
                # a busy pipeline cannot be yanked out from under a generation.
                self._cache.popitem(last=False)
            return entry

    @contextmanager
    def _pipeline(self, model_id: str, kind: str) -> Iterator[Any]:
        """Yield the cached pipeline under its per-pipeline lock.

        Resolution happens before touching the cache (a 404 must not insert
        or evict entries). Creation happens under the per-pipeline lock but
        outside the global cache lock, so loading one model does not block
        requests to other models.
        """
        model_dir = registry.resolve_model_dir(self.settings.models_root, model_id)
        entry = self._entry(model_id, kind)
        with entry.lock:
            if entry.pipeline is None:
                entry.pipeline = self._create_pipeline(model_dir, model_id, kind)
            yield entry.pipeline

    def _create_pipeline(self, model_dir: Path, model_id: str, kind: str) -> Any:
        import openvino_genai as og

        device = self._device_for(model_id)
        if kind == "llm":
            return og.LLMPipeline(model_dir, device)
        if kind == "embedding":
            return og.TextEmbeddingPipeline(model_dir, device)
        raise ValueError(f"Unknown pipeline kind: {kind!r}")

    # -- not implemented in v0.1 (501 contract) ---------------------------------

    def _resolve(self, model_id: str) -> Path:
        return registry.resolve_model_dir(self.settings.models_root, model_id)

    def rerank(
        self,
        model_id: str,
        query: str,
        documents: list[Any],
        top_n: int | None = None,
    ) -> dict:
        self._resolve(model_id)
        raise PipelineNotImplemented("TextRerankPipeline")

    def image_generate(
        self,
        model_id: str,
        prompt: str,
        n: int = 1,
        size: str | None = None,
        response_format: str = "b64_json",
    ) -> dict:
        self._resolve(model_id)
        raise PipelineNotImplemented("Text2ImagePipeline")

    def image_edit(
        self,
        model_id: str,
        prompt: str,
        image: bytes,
        mask: bytes | None,
        n: int = 1,
        size: str | None = None,
        response_format: str = "b64_json",
    ) -> dict:
        self._resolve(model_id)
        if not mask:
            # Zero-length upload counts as "no mask" to avoid silently picking
            # the inpainting path with an empty mask tensor.
            mask = None
        pipeline = "InpaintingPipeline" if mask is not None else "Image2ImagePipeline"
        raise PipelineNotImplemented(pipeline)

    def transcribe(
        self,
        model_id: str,
        audio: bytes,
        language: str | None = None,
        prompt: str | None = None,
        response_format: str = "json",
        temperature: float = 0.0,
    ) -> dict:
        self._resolve(model_id)
        raise PipelineNotImplemented("WhisperPipeline")

    def translate(
        self,
        model_id: str,
        audio: bytes,
        prompt: str | None = None,
        response_format: str = "json",
        temperature: float = 0.0,
    ) -> dict:
        self._resolve(model_id)
        raise PipelineNotImplemented("WhisperPipeline")

    def speech(self, model_id: str, text: str, voice: str | None = None) -> bytes:
        self._resolve(model_id)
        raise PipelineNotImplemented("Text2SpeechPipeline")

    def video_generate(
        self,
        model_id: str,
        prompt: str,
        size: str | None = None,
        seconds: int | None = None,
    ) -> dict:
        self._resolve(model_id)
        raise PipelineNotImplemented("Text2VideoPipeline")
```

    @staticmethod
    def _render_prompt(pipe: Any, messages: list[dict[str, Any]]) -> str:
        """Apply the model's own chat template to OpenAI-style messages."""
        history = [
            {"role": m["role"], "content": _flatten_content(m.get("content"))}
            for m in messages
        ]
        return pipe.get_tokenizer().apply_chat_template(
            history, add_generation_prompt=True
        )

    @staticmethod
    def _generation_config(
        temperature: float | None,
        top_p: float | None,
        max_tokens: int | None,
    ) -> Any:
        import openvino_genai as og

        cfg = og.GenerationConfig()
        cfg.max_new_tokens = max_tokens if max_tokens is not None else 512
        if temperature is not None and temperature > 0:
            cfg.do_sample = True
            cfg.temperature = temperature
            if top_p is not None:
                cfg.top_p = top_p
        # temperature == 0 or unset -> greedy. Deliberate deviation from
        # OpenAI's temperature=1 default: deterministic local inference is
        # more useful.
        return cfg

    @staticmethod
    def _chat_result(result: Any, cfg: Any) -> ChatResult:
        metrics = result.perf_metrics
        completion_tokens = metrics.get_num_generated_tokens()
        prompt_tokens = metrics.get_num_input_tokens()
        finish_reason = (
            "length" if completion_tokens >= cfg.max_new_tokens else "stop"
        )
        return ChatResult(
            text=result.texts[0],
            finish_reason=finish_reason,
            prompt_tokens=prompt_tokens,
            completion_tokens=completion_tokens,
        )

    def _check_chat_supported(self, messages: list[dict[str, Any]]) -> None:
        if messages_have_images(messages):
            raise PipelineNotImplemented("VLMPipeline")
        if self.settings.use_continuous_batching:
            raise PipelineNotImplemented("ContinuousBatchingPipeline")

    def chat(
        self,
        model_id: str,
        messages: list[dict[str, Any]],
        *,
        temperature: float | None = None,
        top_p: float | None = None,
        max_tokens: int | None = None,
    ) -> ChatResult:
        """Blocking non-streaming chat completion."""
        self._check_chat_supported(messages)
        cfg = self._generation_config(temperature, top_p, max_tokens)
        with self._pipeline(model_id, "llm") as pipe:
            prompt = self._render_prompt(pipe, messages)
            result = pipe.generate(prompt, cfg)
        return self._chat_result(result, cfg)

### Task 6: `backend.py` — `chat_stream()`: мост callback → generator с отменой

og зовёт streamer-callback (push), HTTP-стриму нужен итератор (pull). Генерация уходит в worker-поток, callback кладёт дельты в `queue.Queue`, генератор читает. Критическая часть — отмена: клиент закрыл SSE → Starlette закрывает генератор → `GeneratorExit` → callback возвращает `StreamingStatus.STOP` → og прерывает генерацию, устройство и per-pipeline-лок освобождаются. Без этого NPU молотит до конца под локом.

**Files:**
- Modify: `ovinfer/backend.py` (добавить в класс `Backend` сразу после `chat()`)

- [ ] **Step 1: Добавить `chat_stream()` и worker-генератор**

```python
    def chat_stream(
        self,
        model_id: str,
        messages: list[dict[str, Any]],
        *,
        temperature: float | None = None,
        top_p: float | None = None,
        max_tokens: int | None = None,
    ) -> Iterator[str]:
        """Streaming chat: yields str deltas, returns the finish_reason.

        The finish_reason comes back as the generator's return value
        (PEP 380, StopIteration.value). Domain errors (404/501) are raised
        eagerly here, before the transport sends response headers.
        """
        self._check_chat_supported(messages)
        registry.resolve_model_dir(self.settings.models_root, model_id)
        cfg = self._generation_config(temperature, top_p, max_tokens)
        return self._chat_stream_gen(model_id, messages, cfg)

    def _chat_stream_gen(
        self, model_id: str, messages: list[dict[str, Any]], cfg: Any
    ) -> Iterator[str]:
        import openvino_genai as og

        deltas: queue.Queue = queue.Queue()
        cancel = threading.Event()
        outcome: dict[str, Any] = {}

        def on_token(delta: str) -> Any:
            if cancel.is_set():
                return og.StreamingStatus.STOP
            deltas.put(delta)
            return og.StreamingStatus.RUNNING

        def worker() -> None:
            try:
                with self._pipeline(model_id, "llm") as pipe:
                    prompt = self._render_prompt(pipe, messages)
                    result = pipe.generate(prompt, cfg, on_token)
                    outcome["result"] = self._chat_result(result, cfg)
            except BaseException as exc:  # propagated to the consuming thread
                outcome["error"] = exc
            finally:
                deltas.put(_QUEUE_END)

        thread = threading.Thread(
            target=worker, name=f"ovinfer-gen-{model_id}", daemon=True
        )
        thread.start()
        try:
            while True:
                item = deltas.get()
                if item is _QUEUE_END:
                    break
                yield item
        except GeneratorExit:
            # Client disconnected: ask og to stop at the next token, then wait
            # for the worker to release the device and the per-pipeline lock.
            cancel.set()
            thread.join()
            raise
        thread.join()
        if "error" in outcome:
            raise outcome["error"]
        return outcome["result"].finish_reason
```

def _completion_id() -> str:
    return "chatcmpl-" + uuid.uuid4().hex


def _chat_completion_body(model: str, result: ChatResult) -> dict:
    return {
        "id": _completion_id(),
        "object": "chat.completion",
        "created": int(time.time()),
        "model": model,
        "choices": [

            }

def _sse_chat(model: str, deltas: Iterator[str]) -> Iterator[str]:
    """Render backend deltas as chat.completion.chunk SSE events.

    Chunk order is part of the contract: role -> content deltas ->
    finish_reason -> data: [DONE]. The finish_reason is the backend
    generator's return value (StopIteration.value).
    """
    completion_id = _completion_id()
    created = int(time.time())

    def chunk(delta: dict, finish_reason: str | None = None) -> str:
        body = {
            "id": completion_id,
            "object": "chat.completion.chunk",
            "created": created,
            "model": model,
            "choices": [
                {"index": 0, "delta": delta, "finish_reason": finish_reason}
            ],
        }
        return f"data: {json.dumps(body)}\n\n"

    yield chunk({"role": "assistant"})
    finish_reason = "stop"
    try:
        while True:
            try:
                delta = next(deltas)
            except StopIteration as stop:
                if stop.value is not None:
                    finish_reason = stop.value
                break
            yield chunk({"content": delta})
    except Exception as exc:
        # Headers are already out (200): render the error as an SSE event.
        _, body = _map_error(exc)
        yield f"data: {json.dumps(body)}\n\n"
        yield "data: [DONE]\n\n"
        return
    yield chunk({}, finish_reason=finish_reason)
    yield "data: [DONE]\n\n"


def _encode_embedding(vector: list[float], encoding_format: str) -> Any:
    if encoding_format == "base64":
        packed = struct.pack(f"<{len(vector)}f", *vector)
        return base64.b64encode(packed).decode("ascii")
    return vector

```python
"""Console entry point: `ovinfer` with the `server` subcommand.

Subcommand structure leaves room for `ovinfer models ...` in v0.2.
CLI arguments override OVINFER_* env vars (None means "not given here,
fall through to env/defaults" in config.load_settings).
"""

import argparse

from . import __version__


def _parse_listen(raw: str) -> tuple[str, int]:
    host, sep, port = raw.rpartition(":")
    if not sep or not host:
        raise argparse.ArgumentTypeError(f"expected HOST:PORT, got {raw!r}")
    try:
        return host, int(port)
    except ValueError:
        raise argparse.ArgumentTypeError(
            f"port must be an integer, got {port!r}"
        ) from None


def main(argv: list[str] | None = None) -> None:
    parser = argparse.ArgumentParser(
        prog="ovinfer",
        description="OpenVINO GenAI inference server with an OpenAI-compatible API",
    )
    parser.add_argument(
        "--version", action="version", version=f"ovinfer {__version__}"
    )
    sub = parser.add_subparsers(dest="command", required=True)

    server = sub.add_parser("server", help="run the OpenAI-compatible HTTP server")
    server.add_argument(
        "--listen",
        type=_parse_listen,
        default=None,
        metavar="HOST:PORT",
        help="bind address (default 127.0.0.1:8000)",
    )
    server.add_argument(
        "--device", default=None, metavar="DEV",
        help="default inference device (CPU/GPU/NPU)",
    )
    server.add_argument(
        "--device-map", default=None, metavar="MODEL=DEV[,...]",
        help="per-model device overrides",
    )
    server.add_argument(
        "--preload", default=None, metavar="MODEL[:KIND][,...]",
        help="models to warm up at startup (kind: llm|embedding, default llm)",
    )
    server.add_argument(
        "--models-root", default=None, metavar="PATH",
        help="models directory (org/model layout)",
    )

    args = parser.parse_args(argv)
    if args.command == "server":
        _run_server(args)


def _run_server(args: argparse.Namespace) -> None:
    import uvicorn

    from .backend import Backend
    from .config import parse_device_map, parse_preload
    from .errors import ConfigError
    from .server import create_app

    try:
        backend = Backend(
            models_root=args.models_root,
            default_device=args.device,
            device_map=(
                parse_device_map(args.device_map)
                if args.device_map is not None
                else None
            ),
            preload=(
                parse_preload(args.preload) if args.preload is not None else None
            ),
        )
        backend.preload_models()
    except ConfigError as exc:
        raise SystemExit(f"ovinfer: {exc}") from None
    host, port = args.listen or ("127.0.0.1", 8000)
    uvicorn.run(create_app(backend), host=host, port=port)


if __name__ == "__main__":
    main()
```

Self-contained OpenVINO GenAI inference server with an OpenAI-compatible API.

v0.1 implements LLM chat (streaming and non-streaming) and text embeddings;

## Install & run

```bash
uv tool install <path-or-git-url>

```

On machines that need a vendored Intel stack, run under your environment
wrapper (e.g. `openvino-env ovinfer server`).

## Configuration

CLI arguments > `OVINFER_*` environment variables > defaults.

| Setting | CLI | Env | Default |
|---|---|---|---|
| Models directory | `--models-root` | `OVINFER_MODELS_ROOT` | `$XDG_DATA_HOME/ovinfer/models` |
| Default device | `--device` | `OVINFER_DEVICE` | `CPU` |

| Warm-up models | `--preload MODEL[:KIND][,...]` | `OVINFER_PRELOAD` | empty |
| Pipeline cache size | — | `OVINFER_MAX_PIPELINES` | `max(2, len(preload))` |

Models live as `<models-root>/<org>/<model>` OpenVINO IR directories; the
OpenAI `model` field is the `org/model` relative path.
