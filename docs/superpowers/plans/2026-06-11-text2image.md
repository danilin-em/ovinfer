> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wire `og.Text2ImagePipeline` to `POST /v1/images/generations`, replacing the 501 stub with real inference (b64/data-URL PNG output, diffusion knob passthrough).

**Architecture:** Sync core (`Backend.image_generate`) builds the og pipeline (kind `"text2image"`), runs one blocking `generate`, and returns PNG bytes (`ImageResult`); the FastAPI transport parses `size`, forwards optional diffusion fields, and wraps the bytes into the OpenAI images envelope. Dispatch is fixed by route (no request-shape dispatch); a non-diffusion model is rejected with 400 before the pipeline loads.

**Tech Stack:** openvino-genai 2026.2.0, FastAPI, Pillow/numpy (already deps), Python `unittest`.

**Spec:** `docs/superpowers/specs/2026-06-11-text2image-design.md`

def _encode_pngs(tensor: Any) -> list[bytes]:
    """Encode an og image tensor (uint8 RGB, (N,H,W,3) or (H,W,3)) as PNG bytes.

    A single-image result may arrive 3D; it is normalized to a 1-element batch.
    """
    import io

    import numpy as np
    from PIL import Image

    arr = np.array(tensor)
    if arr.ndim == 3:
        arr = arr[np.newaxis, ...]
    pngs: list[bytes] = []
    for i in range(arr.shape[0]):
        buf = io.BytesIO()
        Image.fromarray(arr[i]).save(buf, format="PNG")
        pngs.append(buf.getvalue())
    return pngs
```

    def image_generate(
        self,
        model_id: str,
        prompt: str,
        *,
        n: int = 1,
        width: int | None = None,
        height: int | None = None,
        num_inference_steps: int | None = None,
        guidance_scale: float | None = None,
        negative_prompt: str | None = None,
        seed: int | None = None,
    ) -> ImageResult:
        """Text-to-image generation via og.Text2ImagePipeline.

        Resolution (404) and the diffusion-model check (400 UnsupportedModel)
        run before the pipeline is built. Only explicitly-set diffusion params
        are forwarded so og's own defaults are not clobbered; guidance_scale=0.0
        is forwarded (it is a meaningful value for CFG-distilled turbo models).
        """
        model_dir = registry.resolve_model_dir(self.settings.models_root, model_id)
        if not registry.is_diffusion_model(model_dir):
            raise UnsupportedModel(
                f"model '{model_id}' is not a diffusion model "
                "and cannot generate images"
            )
        kwargs: dict[str, Any] = {"num_images_per_prompt": n}
        if width is not None:
            kwargs["width"] = width
        if height is not None:
            kwargs["height"] = height
        if num_inference_steps is not None:
            kwargs["num_inference_steps"] = num_inference_steps
        if guidance_scale is not None:
            kwargs["guidance_scale"] = guidance_scale
        if negative_prompt is not None:
            kwargs["negative_prompt"] = negative_prompt
        if seed is not None:
            kwargs["rng_seed"] = seed
        with self._pipeline(model_id, "text2image") as pipe:
            tensor = pipe.generate(prompt, **kwargs)
        return ImageResult(images=_encode_pngs(tensor))

def _parse_size(size: str | None) -> tuple[int | None, int | None]:
    """Parse OpenAI 'WIDTHxHEIGHT' into (width, height).

    None or 'auto' -> (None, None): the backend omits width/height and og uses
    the model default. A malformed value raises ValueError (mapped to 400).
    """
    if size is None or size == "auto":
        return None, None
    width, sep, height = size.partition("x")
    if not sep:
        raise ValueError(f"invalid size {size!r}, expected WIDTHxHEIGHT")
    try:
        return int(width), int(height)
    except ValueError:
        raise ValueError(
            f"invalid size {size!r}, expected WIDTHxHEIGHT"
        ) from None
```

def _image_entry(png: bytes, response_format: str) -> dict:
    """One OpenAI images `data` entry. response_format='url' yields a data: URL
    (this server hosts no files); 'b64_json' yields the raw base64 PNG."""
    b64 = base64.b64encode(png).decode("ascii")
    if response_format == "url":
        return {"url": "data:image/png;base64," + b64}
    return {"b64_json": b64}
