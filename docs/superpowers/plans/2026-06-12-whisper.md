# Whisper (ASR) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement batch speech-to-text on `POST /v1/audio/transcriptions` and `POST /v1/audio/translations` via `og.WhisperPipeline`, filling the existing `backend.transcribe`/`translate` 501 stubs.

**Architecture:** New pipeline kind `"whisper"`. A new PyAV-based `_decode_audio` helper turns an uploaded audio file into a mono 16 kHz float32 array (what og wants); `_audio_generate` resolves/validates the model, decodes, and runs one blocking `pipe.generate(samples, **kwargs)` (params passed as generate-kwargs, like the LLM/image pipelines, so unit tests stay og-free). The backend returns a transport-agnostic `TranscriptionResult`; the transport renders it into one of five `response_format`s (json/text/verbose_json/srt/vtt). transcribe vs translate differ only by `task`.

**Tech Stack:** Python 3.12, openvino-genai 2026.2.0, PyAV (`av`, new dep), FastAPI, `unittest`. Tests run with `.venv/bin/python -m unittest`. Dependency management via `uv`.

**Spec:** `docs/superpowers/specs/2026-06-12-whisper-design.md`

**Conventions (verified against the codebase):**
- Tests use `unittest.TestCase`; run with `.venv/bin/python -m unittest tests.test_whisper -v`.
- `og` (openvino_genai) is **never** imported in unit/dispatch/transport tests: pure helpers tested directly, dispatch stubs `_pipeline`, transport stubs the whole backend. `av`/`numpy`/`PIL`/`openvino` are plain deps and may be imported.
- Code/comments/docstrings in **English** (per CLAUDE.md). Russian only in chat.
- All inference on this machine runs through the `openvino-env` wrapper (acceptance only; unit tests don't touch og).

```python
import numpy as np

from ovinfer.backend import _TARGET_SR, _decode_audio
from ovinfer.errors import InvalidAudio


def _wav_bytes(seconds: float = 0.5, sr: int = 16000, channels: int = 1) -> bytes:
    """A valid mono/stereo 16-bit PCM WAV (stdlib only; PyAV reads it)."""
    import math
    import struct
    import wave

    n = int(seconds * sr)
    buf = io.BytesIO()
    with wave.open(buf, "wb") as w:
        w.setnchannels(channels)
        w.setsampwidth(2)
        w.setframerate(sr)
        samples = []
        for t in range(n):
            val = int(0.2 * 32767 * math.sin(2 * math.pi * 440 * t / sr))
            samples.extend([val] * channels)
        w.writeframes(struct.pack("<%dh" % len(samples), *samples))
    return buf.getvalue()


class DecodeAudioTest(unittest.TestCase):
    def test_mono_16k_wav_decodes_to_float32_1d(self):
        out = _decode_audio(_wav_bytes(seconds=0.5, sr=16000))
        self.assertEqual(out.dtype, np.float32)
        self.assertEqual(out.ndim, 1)
        # 0.5 s at 16 kHz ~= 8000 samples (resampler may trim/pad a few).
        self.assertTrue(7000 <= len(out) <= 9000, len(out))
        self.assertLessEqual(float(np.abs(out).max()), 1.0)

    def test_target_sample_rate_is_16k(self):
        self.assertEqual(_TARGET_SR, 16000)

    def test_resamples_8k_up_to_16k(self):
        # 0.5 s at 8 kHz must come back as ~8000 samples (0.5 s at 16 kHz).
        out = _decode_audio(_wav_bytes(seconds=0.5, sr=8000))
        self.assertTrue(7000 <= len(out) <= 9000, len(out))

    def test_stereo_is_downmixed_to_mono_1d(self):
        out = _decode_audio(_wav_bytes(seconds=0.3, sr=16000, channels=2))
        self.assertEqual(out.ndim, 1)
        self.assertTrue(4000 <= len(out) <= 5600, len(out))

            _decode_audio(b"not audio at all")

    def test_empty_bytes_raise_invalid_audio(self):
```

```python
_TARGET_SR = 16000


def _decode_audio(raw: bytes) -> Any:
    """Decode uploaded audio bytes into a mono 16 kHz float32 numpy array.

    og.WhisperPipeline.generate wants raw speech normalized to ~[-1, 1] at
    16 kHz. PyAV (vendored ffmpeg) decodes any container/codec ffmpeg supports
    (wav/mp3/m4a/webm/flac/ogg/opus) in-process; an AudioResampler converts each
    decoded frame to mono float ('flt' is already in [-1, 1]) at 16 kHz, and the
    frames are concatenated. Broken/unsupported input raises InvalidAudio (400).
    """
    import io

    import av
    import numpy as np

    try:
        container = av.open(io.BytesIO(raw))
    except (av.error.FFmpegError, OSError, ValueError) as exc:
        raise InvalidAudio(f"could not open audio: {exc}") from exc
    try:
        if not container.streams.audio:
            raise InvalidAudio("file has no audio stream")
        stream = container.streams.audio[0]
        resampler = av.AudioResampler(format="flt", layout="mono", rate=_TARGET_SR)
        chunks: list[Any] = []
        for frame in container.decode(stream):
            for resampled in resampler.resample(frame):
                chunks.append(resampled.to_ndarray().reshape(-1))
        for resampled in resampler.resample(None):  # flush the resampler tail
            chunks.append(resampled.to_ndarray().reshape(-1))
    except InvalidAudio:
        raise
    except (av.error.FFmpegError, OSError, ValueError) as exc:
        raise InvalidAudio(f"could not decode audio: {exc}") from exc
    finally:
        container.close()
    if not chunks:
        raise InvalidAudio("audio file contains no decodable samples")
    return np.concatenate(chunks).astype(np.float32)
```

In `ovinfer/backend.py`, replace the two stub methods (`backend.py:732-753`, the `transcribe` and `translate` defs that call `self._resolve` and raise `PipelineNotImplemented("WhisperPipeline")`) with:

```python
    # -- audio transcription ------------------------------------------------------

    _TIMESTAMP_FORMATS = ("verbose_json", "srt", "vtt")

    def transcribe(
        self,
        model_id: str,
        audio: bytes,
        language: str | None = None,
        prompt: str | None = None,
        response_format: str = "json",
        temperature: float = 0.0,
    ) -> TranscriptionResult:
        """Speech-to-text in the audio's own language via og.WhisperPipeline."""
        return self._audio_generate(
            model_id, audio, "transcribe", language, prompt,
            response_format, temperature,
        )

    def translate(
        self,
        model_id: str,
        audio: bytes,
        prompt: str | None = None,
        response_format: str = "json",
        temperature: float = 0.0,
    ) -> TranscriptionResult:
        """Speech-to-English translation via og.WhisperPipeline (task=translate)."""
        return self._audio_generate(
            model_id, audio, "translate", None, prompt,
            response_format, temperature,
        )

    def _audio_generate(
        self,
        model_id: str,
        audio: bytes,
        task: str,
        language: str | None,
        prompt: str | None,
        response_format: str,
        temperature: float,
    ) -> TranscriptionResult:
        """Shared core for transcribe/translate.

        Check order tracks cost (like image_edit): model resolution (404) first;
        the Whisper-model check (400 UnsupportedModel) before any decode; the
        audio is decoded (400 InvalidAudio on bad bytes) before the expensive
        pipeline build, so a broken upload 400s without loading the model.
        Generation params go as generate(**kwargs) — og maps them onto
        WhisperGenerationConfig fields, matching the LLM/image pipelines — and
        only explicitly-set params are forwarded. Timestamps (og chunks) are
        requested only for the formats that render them.
        """
        model_dir = registry.resolve_model_dir(self.settings.models_root, model_id)
        if not registry.is_whisper_model(model_dir):
            raise UnsupportedModel(
                f"model '{model_id}' is not a Whisper model "
                "and cannot transcribe audio"
            )
        samples = _decode_audio(audio)
        want_timestamps = response_format in self._TIMESTAMP_FORMATS
        kwargs: dict[str, Any] = {"task": task}
        if language is not None:
            kwargs["language"] = f"<|{language}|>"
        if prompt:
            kwargs["initial_prompt"] = prompt
        if temperature and temperature > 0:
            kwargs["do_sample"] = True
            kwargs["temperature"] = temperature
        if want_timestamps:
            kwargs["return_timestamps"] = True
        with self._pipeline(model_id, "whisper") as pipe:
            result = pipe.generate(samples, **kwargs)
        text = result.texts[0].strip() if result.texts else ""
        segments = None
        if want_timestamps:
            segments = [
                Segment(start=float(c.start_ts), end=float(c.end_ts), text=c.text)
                for c in (result.chunks or [])
            ]
        return TranscriptionResult(
            text=text,
            language=_strip_lang_token(getattr(result, "language", None)),
            duration=len(samples) / _TARGET_SR,
            segments=segments,
        )
```

```python
# --- audio transcription rendering ----------------------------------------------

def _format_timestamp(seconds: float, sep: str) -> str:
    """Seconds -> 'HH:MM:SS{sep}mmm'. sep is ',' for SRT, '.' for VTT."""
    if seconds < 0:
        seconds = 0.0
    millis = int(round(seconds * 1000))
    hours, millis = divmod(millis, 3_600_000)
    minutes, millis = divmod(millis, 60_000)
    secs, millis = divmod(millis, 1000)
    return f"{hours:02d}:{minutes:02d}:{secs:02d}{sep}{millis:03d}"


def _segments_to_srt(segments: list) -> str:
    """Render segments as SubRip (SRT): 1-based index, comma decimal separator."""
    blocks = []
    for i, seg in enumerate(segments, start=1):
        start = _format_timestamp(seg.start, ",")
        end = _format_timestamp(seg.end, ",")
        blocks.append(f"{i}\n{start} --> {end}\n{seg.text.strip()}\n")
    return "\n".join(blocks)


def _segments_to_vtt(segments: list) -> str:
    """Render segments as WebVTT: 'WEBVTT' header, dot decimal separator."""
    blocks = ["WEBVTT", ""]
    for seg in segments:
        start = _format_timestamp(seg.start, ".")
        end = _format_timestamp(seg.end, ".")
        blocks.append(f"{start} --> {end}\n{seg.text.strip()}\n")
    return "\n".join(blocks)


def _verbose_json_body(result: Any, task: str) -> dict:
    """OpenAI verbose_json body. Analytics fields og does not provide
    (avg_logprob, no_speech_prob, ...) are omitted; clients read start/end/text."""
    segments = result.segments or []
    return {
        "task": task,
        "language": result.language,
        "duration": result.duration,
        "text": result.text,
        "segments": [
            {"id": i, "seek": 0, "start": seg.start, "end": seg.end,
             "text": seg.text}
            for i, seg in enumerate(segments)
        ],
    }


def _render_transcription(result: Any, response_format: str, task: str) -> Any:
    """Render a TranscriptionResult per response_format. json/verbose_json return
    a dict (JSON); text/srt/vtt return a plain-text Response with the right
    media type."""
    if response_format == "text":
        return PlainTextResponse(result.text)
    if response_format == "srt":
        return Response(
            content=_segments_to_srt(result.segments or []),
            media_type="application/x-subrip",
        )
    if response_format == "vtt":
        return Response(
            content=_segments_to_vtt(result.segments or []),
            media_type="text/vtt",
        )
    if response_format == "verbose_json":
        return _verbose_json_body(result, task)
    return {"text": result.text}
```
